def znajdz_pola(tablica, p):
    pola = []
    for element_1 in tablica:
        for element_2 in tablica:
            if element_1 == element_2:
                continue
            pole = element_1 * element_2
            if pole not in pola and pole % p != 0:
                pola.append(pole)
    return pola
def znajdz_najwieksze_pole(tablica, p):
    pola = znajdz_pola(tablica, p)
    najwieksze_pole = 0
    for pole in pola:
        if pole > najwieksze_pole:
            najwieksze_pole = pole
    return najwieksze_pole
