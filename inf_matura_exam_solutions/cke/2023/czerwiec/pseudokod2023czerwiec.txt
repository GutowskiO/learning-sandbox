a = 433
b = 352
def znajdz_cyfry(liczba):
    cyfry = []
    while liczba > 0:
        n = liczba % 10
        cyfry.append(n)
        liczba //= 10
    return cyfry[::-1]    

def czy_bardzo_przyjaciolki(cyfry_a,cyfry_b):
    suma_a = 0
    suma_b = 0
    for a in range(len(cyfry_a)):
        suma_a += cyfry_a[a]
    for b in range(len(cyfry_b)):
        suma_b += cyfry_b[b]
        
    if suma_a == suma_b and (cyfry_a[0] == cyfry_b[len(cyfry_b)-1] or cyfry_b[0] == cyfry_a[len(cyfry_a)-1]):
        return True
    return False
    
wynik = czy_bardzo_przyjaciolki(znajdz_cyfry(a),znajdz_cyfry(b))
print(wynik)
