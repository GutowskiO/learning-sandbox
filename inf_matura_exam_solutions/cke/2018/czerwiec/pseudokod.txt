def kwadrat(x):
    return x*x

def f(n):
    if n == 1 or n == 2:
        return 1
    if n%2 ==0:
        return kwadrat(f(n//2 +1)) - kwadrat(f(n//2 -1))
    else:
        return kwadrat(f((n+1)//2))+ kwadrat(f((n+1)//2 -1))

print(f(4))
