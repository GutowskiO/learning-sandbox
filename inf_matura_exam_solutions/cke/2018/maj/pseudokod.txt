n = 4
X = [1, 3, 2, -2]
Y = [3, 4, 1, 2]

def znajdz_najbardziej_na_lewo_x(x):
    wynik = x[0]
    miejsce_wyniku = 0
    for a in range(len(x)):
        if wynik > x[a]:
            wynik = x[a]
            miejsce_wyniku = a
    return miejsce_wyniku
miejsce_wyniku = znajdz_najbardziej_na_lewo_x(X)
x = X[miejsce_wyniku]
y = Y[miejsce_wyniku]
print(x)
print(y)
