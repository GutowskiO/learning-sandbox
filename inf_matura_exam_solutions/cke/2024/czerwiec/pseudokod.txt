def czy_nieliczna(k):
    czy_ostatnia_1 = False
    for a in range(k, -1,-1):
        potega = 1
        for b in range(a):
            potega  = potega*2
        if potega<=k:
            if czy_ostatnia_1:
                return False
            czy_ostatnia_1 = True
            k -= potega
            continue
        czy_ostatnia_1 = False
    return True

print(czy_nieliczna(259))
