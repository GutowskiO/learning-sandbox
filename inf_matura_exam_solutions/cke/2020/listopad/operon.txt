def dec_to_bin(n):
    potega = 0
    binarna = ""
    while 2 ** potega <= n:
        potega += 1
    potega -= 1    
    
    while potega >= 0:
        
        if 2 ** potega <= n:
            binarna = binarna + "1"
            n -= 2 ** potega
            
        else:
            if len(binarna) == 0:
                potega -= 1
                continue
            binarna = binarna + "0"
        potega -= 1   
    return binarna
def odwroc(string):
    odwrocony = ""
    for a in range(len(string)):
        odwrocony = odwrocony + string[len(string)-a-1]
    return odwrocony
def czy_palindrom(n):
    n = str(n)
    if odwroc(n) == n:
        return True
    return False
